// КОНТЕКСТ: Форма справочника "ФизическиеЛица"
// НАВЫКИ: &НаКлиенте, обработка событий, работа со строками, Сообщить()

&НаКлиенте
Процедура ПередЗаписью(Отказ, ПараметрыЗаписи)
    // Проверка обязательных полей перед формированием ФИО
    ТекстСообщения = "";
    
    Если ПустаяСтрока(Объект.Фамилия) Тогда
        ТекстСообщения = ТекстСообщения + "Не заполнено поле Фамилия;" + Символы.ПС;
    КонецЕсли;
    
    Если ПустаяСтрока(Объект.Имя) Тогда
        ТекстСообщения = ТекстСообщения + "Не заполнено поле Имя;" + Символы.ПС;
    КонецЕсли;

    Если НЕ ПустаяСтрока(ТекстСообщения) Тогда
        Сообщить(ТекстСообщения);
        Отказ = Истина; // Блокируем запись объекта
        Возврат;
    КонецЕсли;

    // Формирование наименования: Фамилия И.О.
    Объект.Наименование = Объект.Фамилия + " " 
        + ВРег(Лев(Объект.Имя, 1)) + ". " 
        + ВРег(Лев(Объект.Отчество, 1)) + ".";
        
КонецПроцедуры
